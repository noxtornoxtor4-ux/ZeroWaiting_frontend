# Custom Hall Layouts Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** End-to-end support for custom cinema hall layouts — grid-based admin constructor, atomic layout API, full rewrite of the client seat map — replacing the current rectangular-only seat model.

**Architecture:** Backend keeps `Seat` as first-class DB rows (`SeatHold` / `BookingSeat` / `Ticket` FKs stay intact) but switches positioning from `rowLabel`/`positionX/Y` to integer `gridRow`/`gridCol`/`widthCells`. Hall gains `layoutRows`/`layoutCols`. A single atomic `PUT /halls/:id/layout` endpoint replaces all per-seat CRUD. Frontend gets a grid-painter editor in admin and a CSS-grid-based seat map on client.

**Tech Stack:**

- Backend: NestJS, Prisma, PostgreSQL, Vitest (`~/Desktop/ZeroWaiting_backend`)

ALTER TABLE "Hall" ADD COLUMN "layoutRows" INTEGER;
ALTER TABLE "Hall" ADD COLUMN "layoutCols" INTEGER;

UPDATE "Hall" h SET
  "layoutRows" = COALESCE((SELECT MAX("rowNumber") FROM "Seat" s WHERE s."hallId" = h.id), 1),
  "layoutCols" = COALESCE((SELECT MAX("seatNumber") FROM "Seat" s WHERE s."hallId" = h.id), 1);

ALTER TABLE "Hall" ALTER COLUMN "layoutRows" SET NOT NULL;
ALTER TABLE "Hall" ALTER COLUMN "layoutCols" SET NOT NULL;

-- Seat: add new columns nullable, backfill from existing rowNumber/seatNumber, then set NOT NULL
ALTER TABLE "Seat" ADD COLUMN "gridRow" INTEGER;
ALTER TABLE "Seat" ADD COLUMN "gridCol" INTEGER;
ALTER TABLE "Seat" ADD COLUMN "widthCells" INTEGER NOT NULL DEFAULT 1;

UPDATE "Seat" SET "gridRow" = "rowNumber" - 1, "gridCol" = "seatNumber" - 1;

ALTER TABLE "Seat" ALTER COLUMN "gridRow" SET NOT NULL;
ALTER TABLE "Seat" ALTER COLUMN "gridCol" SET NOT NULL;

-- Drop obsolete columns
ALTER TABLE "Seat" DROP COLUMN "rowLabel";
ALTER TABLE "Seat" DROP COLUMN "positionX";
ALTER TABLE "Seat" DROP COLUMN "positionY";

-- Add new enum values
ALTER TYPE "SeatType" ADD VALUE 'LOVE_SEAT';
ALTER TYPE "SeatType" ADD VALUE 'RECLINER';
ALTER TYPE "SeatType" ADD VALUE 'DIRECTOR';

-- New unique constraint on grid position

```typescript
import { SeatType } from '@prisma/client';

export interface SeatLayoutInput {
	gridRow: number;
	gridCol: number;
	widthCells: number;
	type: SeatType;
	section?: string | null;
}

export interface SeatLayoutLabeled extends SeatLayoutInput {
	rowNumber: number;
	seatNumber: number;
}

export interface LayoutValidationError {
	code: 'OUT_OF_BOUNDS' | 'OVERLAP' | 'INVALID_WIDTH';
	message: string;
	seat?: Pick<SeatLayoutInput, 'gridRow' | 'gridCol'>;
}

export const computeLabels = (
	seats: SeatLayoutInput[]
): SeatLayoutLabeled[] => {
	const populatedRows = [...new Set(seats.map((s) => s.gridRow))].sort(
		(a, b) => a - b
	);
	const rowToNumber = new Map<number, number>();
	populatedRows.forEach((row, idx) => rowToNumber.set(row, idx + 1));

	const byRow = new Map<number, SeatLayoutInput[]>();
	for (const seat of seats) {
		const list = byRow.get(seat.gridRow) ?? [];
		list.push(seat);
		byRow.set(seat.gridRow, list);
	}

	const result: SeatLayoutLabeled[] = [];
	for (const [gridRow, rowSeats] of byRow) {
		const sorted = [...rowSeats].sort((a, b) => a.gridCol - b.gridCol);
		sorted.forEach((seat, idx) => {
			result.push({
				...seat,
				rowNumber: rowToNumber.get(gridRow)!,
				seatNumber: idx + 1
			});
		});
	}
	return result;
};

export const validateLayout = (payload: {
	layoutRows: number;
	layoutCols: number;
	seats: SeatLayoutInput[];
}): LayoutValidationError[] => {
	const errors: LayoutValidationError[] = [];
	const occupied = new Map<string, SeatLayoutInput>();

	for (const seat of payload.seats) {
		if (seat.widthCells < 1 || seat.widthCells > 2) {
			errors.push({
				code: 'INVALID_WIDTH',
				message: `widthCells must be 1 or 2`,
				seat
			});
			continue;
		}
		if (
			seat.gridRow < 0 ||
			seat.gridRow >= payload.layoutRows ||
			seat.gridCol < 0 ||
			seat.gridCol + seat.widthCells - 1 >= payload.layoutCols
		) {
			errors.push({
				code: 'OUT_OF_BOUNDS',
				message: `seat at (${seat.gridRow}, ${seat.gridCol}) out of ${payload.layoutRows}x${payload.layoutCols}`,
				seat
			});
			continue;
		}
		for (let dc = 0; dc < seat.widthCells; dc++) {
			const key = `${seat.gridRow}:${seat.gridCol + dc}`;
			if (occupied.has(key)) {
				errors.push({
					code: 'OVERLAP',
					message: `two seats at cell (${seat.gridRow}, ${seat.gridCol + dc})`,
					seat
				});
			}
			occupied.set(key, seat);
		}
	}
	return errors;
};
```

		seatNumber: number;
		type: SeatType;
		section: string | null;
		isActive: boolean;
	}>;
}

const ACTIVE_BOOKING_STATUSES: Prisma.BookingStatus[] = [
	'PENDING',
	'CONFIRMED'
];

@Injectable()
export class HallLayoutService {
	constructor(private readonly prisma: PrismaService) {}

	async getLayout(hallId: string): Promise<HallLayoutResult> {
		const hall = await this.prisma.hall.findUnique({ where: { id: hallId } });
		if (!hall) throw new NotFoundException(`Hall ${hallId} not found`);
		const seats = await this.prisma.seat.findMany({
			where: { hallId },
			orderBy: [{ gridRow: 'asc' }, { gridCol: 'asc' }]
		});
		return {
			hallId,
			layoutRows: hall.layoutRows,
			layoutCols: hall.layoutCols,
			seats
		};
	}

	async saveLayout(
		hallId: string,
		payload: SaveLayoutPayload
	): Promise<HallLayoutResult> {
		const hall = await this.prisma.hall.findUnique({ where: { id: hallId } });
		if (!hall) throw new NotFoundException(`Hall ${hallId} not found`);

		const errors = validateLayout(payload);
		if (errors.length) {
			throw new BadRequestException({ message: 'Invalid layout', errors });
		}

		const labeled = computeLabels(payload.seats);

		return this.prisma.$transaction(async (tx) => {
			const current = await tx.seat.findMany({ where: { hallId } });
			const currentByCell = new Map(
				current.map((s) => [`${s.gridRow}:${s.gridCol}`, s])
			);
			const payloadCells = new Set(
				labeled.map((s) => `${s.gridRow}:${s.gridCol}`)
			);

			const toDelete = current.filter(
				(s) => !payloadCells.has(`${s.gridRow}:${s.gridCol}`)
			);
			if (toDelete.length) {
				const seatIds = toDelete.map((s) => s.id);
				const [holds, bookings, tickets] = await Promise.all([
					tx.seatHold.findMany({ where: { seatId: { in: seatIds } } }),
					tx.bookingSeat.findMany({
						where: {
							seatId: { in: seatIds },
							booking: { status: { in: ACTIVE_BOOKING_STATUSES } }
						},
						include: { booking: true }
					}),
					tx.ticket.findMany({ where: { seatId: { in: seatIds } } })
				]);
				if (holds.length || bookings.length || tickets.length) {
					throw new ConflictException({
						message: 'Cannot remove seats with active bookings or holds',
						conflicts: [
							...new Set(
								[...holds, ...bookings, ...tickets].map((x) => x.seatId)
							)
						]
					});
				}
				await tx.seat.deleteMany({ where: { id: { in: seatIds } } });
			}

			for (const seat of labeled) {
				const key = `${seat.gridRow}:${seat.gridCol}`;
				const existing = currentByCell.get(key);
				if (existing) {
					await tx.seat.update({
						where: { id: existing.id },
						data: {
							type: seat.type,
							rowNumber: seat.rowNumber,
							seatNumber: seat.seatNumber,
							widthCells: seat.widthCells,
							section: seat.section ?? null
						}
					});
				} else {
					await tx.seat.create({
						data: {
							hallId,
							gridRow: seat.gridRow,
							gridCol: seat.gridCol,
							widthCells: seat.widthCells,
							rowNumber: seat.rowNumber,
							seatNumber: seat.seatNumber,
							type: seat.type,
							section: seat.section ?? null
						}
					});
				}
			}

			const newCount = await tx.seat.count({ where: { hallId } });
			await tx.hall.update({
				where: { id: hallId },
				data: {
					layoutRows: payload.layoutRows,
					layoutCols: payload.layoutCols,
					capacity: newCount
				}
			});

			return this.getLayout(hallId);
		});
	}
}

import type { SeatCell } from './layout-state.svelte';
import type { SeatType } from '@/api/model';

export const computeClientLabels = (
	cells: SeatCell[]
): Map<string, { rowNumber: number; seatNumber: number }> => {
	const rows = [...new Set(cells.map((c) => c.gridRow))].sort((a, b) => a - b);
	const rowIndex = new Map(rows.map((r, i) => [r, i + 1]));
	const out = new Map<string, { rowNumber: number; seatNumber: number }>();
	const byRow = new Map<number, SeatCell[]>();
	for (const c of cells) {
		(byRow.get(c.gridRow) ?? byRow.set(c.gridRow, []).get(c.gridRow)!).push(c);
	}
	for (const [row, rowCells] of byRow) {
		const sorted = [...rowCells].sort((a, b) => a.gridCol - b.gridCol);
		sorted.forEach((c, i) => {
			out.set(`${c.gridRow}:${c.gridCol}`, {
				rowNumber: rowIndex.get(row)!,
				seatNumber: i + 1
			});
		});
	}
	return out;
};

type Counts = Record<SeatType, number> & { total: number };

export const countSeatsByType = (cells: SeatCell[]): Counts => {
	const base: Counts = {
		STANDARD: 0,
		VIP: 0,
		LOVE_SEAT: 0,
		RECLINER: 0,
		WHEELCHAIR: 0,
		DIRECTOR: 0,
		total: 0
	};
	for (const c of cells) {
		base[c.type] = (base[c.type] ?? 0) + 1;
		base.total += 1;
	}
	return base;
};

```typescript
import type { SeatType } from '@/api/model';

export interface SeatCell {
	gridRow: number;
	gridCol: number;
	widthCells: 1 | 2;
	type: SeatType;
}

type PaintTool = SeatType | 'ERASER';

export interface LayoutSnapshot {
	layoutRows: number;
	layoutCols: number;
	cells: SeatCell[];
}

const UNDO_LIMIT = 20;

const cloneSnapshot = (s: LayoutSnapshot): LayoutSnapshot => ({
	layoutRows: s.layoutRows,
	layoutCols: s.layoutCols,
	cells: s.cells.map((c) => ({ ...c }))
});

export const createLayoutState = (initial: LayoutSnapshot) => {
	let current = $state<LayoutSnapshot>(cloneSnapshot(initial));
	let undoStack: LayoutSnapshot[] = [];
	let redoStack: LayoutSnapshot[] = [];
	let activeTool = $state<PaintTool>('STANDARD');

	const pushHistory = () => {
		undoStack.push(cloneSnapshot(current));
		if (undoStack.length > UNDO_LIMIT) undoStack.shift();
		redoStack = [];
	};

	const cellAt = (row: number, col: number): SeatCell | undefined =>
		current.cells.find(
			(c) =>
				c.gridRow === row && col >= c.gridCol && col < c.gridCol + c.widthCells
		);

	const setCell = (row: number, col: number, tool: PaintTool) => {
		pushHistory();
		const occupying = cellAt(row, col);
		const filtered = current.cells.filter((c) => c !== occupying);

		if (tool === 'ERASER') {
			current = { ...current, cells: filtered };
			return;
		}

		const widthCells = tool === 'LOVE_SEAT' ? 2 : 1;
		if (col + widthCells > current.layoutCols) return; // out of bounds, no-op
		if (widthCells === 2) {
			const rightNeighbor = cellAt(row, col + 1);
			if (rightNeighbor && rightNeighbor !== occupying) return; // right cell busy
		}

		current = {
			...current,
			cells: [
				...filtered,
				{ gridRow: row, gridCol: col, widthCells, type: tool }
			]
		};
	};

	const resize = (rows: number, cols: number) => {
		pushHistory();
		const inBounds = current.cells.filter(
			(c) => c.gridRow < rows && c.gridCol + c.widthCells <= cols
		);
		current = { layoutRows: rows, layoutCols: cols, cells: inBounds };
	};

	const autoFillStandard = () => {
		pushHistory();
		const cells: SeatCell[] = [];
		for (let r = 0; r < current.layoutRows; r++) {
			for (let c = 0; c < current.layoutCols; c++) {
				cells.push({ gridRow: r, gridCol: c, widthCells: 1, type: 'STANDARD' });
			}
		}
		current = { ...current, cells };
	};

	const clear = () => {
		pushHistory();
		current = { ...current, cells: [] };
	};

	const undo = () => {
		const prev = undoStack.pop();
		if (prev) {
			redoStack.push(cloneSnapshot(current));
			current = prev;
		}
	};

	const redo = () => {
		const next = redoStack.pop();
		if (next) {
			undoStack.push(cloneSnapshot(current));
			current = next;
		}
	};

	return {
		get snapshot() {
			return current;
		},
		get activeTool() {
			return activeTool;
		},
		setActiveTool(t: PaintTool) {
			activeTool = t;
		},
		cellAt,
		setCell,
		resize,
		autoFillStandard,
		clear,
		undo,
		redo,
		canUndo: () => undoStack.length > 0,
		canRedo: () => redoStack.length > 0
	};
};

export type LayoutState = ReturnType<typeof createLayoutState>;
```

	});

		const seat = seats.find((s) => s.gridRow === gridRow);

		s.status === 'BOOKED' || s.status === 'HELD';
</script>

						{#each Array.from({ length: hall.layoutCols }) as _, c}
							{@const seat = byCell.get(`${r}:${c}`)}
							{@const isLeft = seat && seat.gridCol === c}
							{#if !seat}
								<div class="gap"></div>
							{:else if isLeft}
								<!-- seat owner cell; the right-half of love-seat (seat && !isLeft) is skipped because the span-2 button above already covers it -->
								<button

									class:held={seat.status === 'HELD'}
									class:held_by_you={seat.status === 'HELD_BY_YOU'}
									data-type={seat.type}
									style:grid-column="span {seat.widthCells}"
									disabled={isDisabled(seat)}
									aria-label={`Ряд ${seat.rowNumber}, Место ${seat.seatNumber}`}
									onclick={() => onToggle(seat.id)}
								>
									{seat.seatNumber}
								</button>
							{/if}
						{/each}
					</div>
					<span class="row_label">{displayRowNumber(r)}</span>
				</div>
			{/if}
		{/each}
