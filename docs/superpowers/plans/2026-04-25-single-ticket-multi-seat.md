# Single Ticket per Booking (Multi-Seat QR) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace «N Tickets per N seats» with «1 Ticket per 1 Booking» + single QR-URL. Add public status page `/t/[qrCode]` and STAFF-only scanner page `/scanner` with a shortcut in the header.

**Architecture:** `Booking 1:1 Ticket` (pivot `BookingSeat` stays as source of truth for seats). QR payload is a public URL; scanner (STAFF+) parses `qrCode` from it and calls the existing validate/scan endpoints; customers get a public read-only status page. One Prisma migration, schema-greenfield (no data compat).

**Tech Stack:**
- Backend: NestJS 11 + Prisma 7 + PostgreSQL + Vitest + Nodemailer + `qrcode`
- Frontend: SvelteKit 2 (Svelte 5 runes) + TanStack Svelte Query + Orval + svelte-i18n + Vitest + `@zxing/browser`
- Spec: `docs/superpowers/specs/2026-04-25-single-ticket-multi-seat-design.md`

**Scope check:** Single feature, cross-repo but logically one change. Not decomposable into smaller specs — the backend rewrite is meaningless without the frontend update, and vice versa.

**Release model:** Greenfield. Backend lands first (schema + services + email), frontend regenerates Orval and updates UI. Coordinated deploy (local dev → both repos together).

```typescript
  const ticket = await this.prisma.ticket.findUnique({
```

```typescript
    startTime: screening.startTime.toISOString(),
```

```typescript
export interface SeatLabel {
  rowNumber: number;
  seatNumber: number;
}

export function formatSeats(seats: SeatLabel[]): string {
  if (seats.length === 0) return '';
  const byRow = new Map<number, number[]>();
  for (const s of seats) {
    if (!byRow.has(s.rowNumber)) byRow.set(s.rowNumber, []);
    byRow.get(s.rowNumber)!.push(s.seatNumber);
  }
  const parts: string[] = [];
  for (const row of [...byRow.keys()].sort((a, b) => a - b)) {
    const seatsInRow = byRow.get(row)!.sort((a, b) => a - b);
    const word = seatsInRow.length === 1 ? 'место' : 'места';
    parts.push(`Ряд ${row}: ${word} ${seatsInRow.join(', ')}`);
  }
  return parts.join(' · ');
}
```

```typescript
interface SeatLike {
  rowNumber: number;
  seatNumber: number;
}

export interface GroupedRow {
  row: number;
  seats: number[];
}

export const groupSeatsByRow = (seats: SeatLike[]): GroupedRow[] => {
  const byRow = new Map<number, number[]>();
  for (const s of seats) {
    const arr = byRow.get(s.rowNumber) ?? [];
    arr.push(s.seatNumber);
    byRow.set(s.rowNumber, arr);
  }
  return [...byRow.entries()]
    .sort(([a], [b]) => a - b)
    .map(([row, seats]) => ({ row, seats: seats.sort((a, b) => a - b) }));
};
```

```typescript
const TICKET_PATH_REGEX = /\/t\/([^/?#]+)$/;

export const parseTicketQr = (raw: string): string | null => {
```

```svelte
<script lang="ts">
  import { onDestroy } from 'svelte';
  import { BrowserMultiFormatReader } from '@zxing/browser';
  import { _ } from 'svelte-i18n';
  import { crmQueryApi } from '@/api/endpoints';
  import { parseTicketQr } from '@/lib/utils/parse-ticket-url';
  import ScannerResult from './components/ScannerResult.svelte';

  type Phase = 'scanning' | 'result' | 'error';

  let videoEl: HTMLVideoElement;
  let phase = $state<Phase>('scanning');
  let errorMsg = $state('');
  let currentQrCode = $state<string | null>(null);

  const reader = new BrowserMultiFormatReader();
  let stop: (() => void) | null = null;

  const startScanning = async () => {
    try {
      const controls = await reader.decodeFromVideoDevice(
        undefined,
        videoEl,
        (result) => {
          if (!result) return;
          const qrCode = parseTicketQr(result.getText());
          if (!qrCode) {
            errorMsg = $_('scanner.unknownFormat');
            return;
          }
          currentQrCode = qrCode;
          phase = 'result';
          stop?.();
        },
      );
      stop = () => controls.stop();
    } catch {
      phase = 'error';
      errorMsg = $_('scanner.noCamera');
    }
  };

  $effect(() => {
    startScanning();
  });
  onDestroy(() => stop?.());

  const ticketQuery = $derived(
    currentQrCode
      ? crmQueryApi.createGetTicketsByQrCodeValidateV1(() => currentQrCode!)
      : null,
  );
  const scanMutation = crmQueryApi.createPatchTicketsByQrCodeScanV1Mutation();

  const markUsed = async () => {
    if (!currentQrCode) return;
    try {
      await scanMutation.mutateAsync({ qrCode: currentQrCode });
    } catch (e) {
      errorMsg = $_('scanner.networkError');
    }
  };

  const next = () => {
    currentQrCode = null;
    phase = 'scanning';
    errorMsg = '';
    startScanning();
  };
</script>

<div class="scanner_viewport">
  {#if phase === 'scanning'}
    <video bind:this={videoEl} autoplay playsinline muted></video>
    <p class="hint">{$_('scanner.scanHint')}</p>
  {:else if phase === 'result' && ticketQuery}
    {#if ticketQuery.data}
      <ScannerResult
        ticket={ticketQuery.data}
        pending={scanMutation.isPending}
        onMarkUsed={markUsed}
        onNext={next}
      />
    {:else if ticketQuery.isLoading}
      <p class="hint">…</p>
    {:else}
      <p class="hint">{$_('scanner.unknownFormat')}</p>
      <button type="button" onclick={next}>{$_('scanner.next')}</button>
    {/if}
  {:else}
    <p class="hint error">{errorMsg}</p>
    <button type="button" onclick={next}>{$_('scanner.next')}</button>
  {/if}
</div>

<style lang="scss">
  .scanner_viewport {
    flex: 1;
    display: flex;
    flex-direction: column;
    align-items: stretch;
    video { width: 100%; height: 60vh; object-fit: cover; background: #000; }

    .error { color: #ef4444; }

      border-radius: 8px;
      border: 1px solid #2d2d3d;
      background: transparent;

    }

</style>
```
